--table SETUP

CREATE TABLE trainer (tid INT PRIMARY KEY,tname string);
CREATE TABLE pokemon (pid INT PRIMARY KEY,pname string,atk int,def int, trainer_id INT unique references trainer);


CREATE TABLE student (sid INT PRIMARY KEY,sname string,grade INT);

CREATE TABLE course (cid INT PRIMARY KEY,course_name string);

CREATE TABLE student_course (student_id INT references student,course_id INT references course);


--  RBAC SETUP

CREATE ROLE wasi with secret 'omar';
CREATE ROLE saif with secret 'saif';

GRANT SELECT, delete ON student TO wasi;
GRANT SELECT, insert, update ON pokemon TO saif;
